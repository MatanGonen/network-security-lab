! OSPF — Underlay for iBGP
router ospf 1
 router-id 3.3.3.3
 network 3.3.3.3 0.0.0.0 area 0
 network 10.1.1.0 0.0.0.3 area 0
 network 10.0.0.0 0.0.0.3 area 0

! OSPFv3 for IPv6 underlay
ipv6 router ospf 1
 router-id 3.3.3.3

! BGP — iBGP to R1 only
router bgp 65001
 bgp router-id 3.3.3.3
 bgp log-neighbor-changes
 no auto-summary
 
 neighbor 1.1.1.1 remote-as 65001
 neighbor 1.1.1.1 description IBGP_TO_R1
 neighbor 1.1.1.1 update-source Loopback0
 neighbor 1.1.1.1 next-hop-self
 
 network 3.3.3.3 mask 255.255.255.255
 network 10.0.0.0 mask 255.255.255.252
 
 redistribute ospf 1 route-map OSPF_TO_BGP

 
 ! Activate neighbor in IPv6 AF
 address-family ipv6 unicast
  neighbor 2001:DB8::1 remote-as 65001
  neighbor 2001:DB8::1 update-source Loopback0
  neighbor 2001:DB8::1 next-hop-self
  neighbor 2001:DB8::1 activate
  
  network 2001:DB8::3/128
 exit-address-family


ip prefix-list OSPF_ROUTES permit 10.10.10.0/30
route-map OSPF_TO_BGP permit 10
 match ip address prefix-list OSPF_ROUTES

end
